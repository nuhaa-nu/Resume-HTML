<script>
const statusEl = document.getElementById('status');
const contentEl = document.getElementById('content');

// WMO weather code -> icon + description
const WMO = {
  0:['☀️','Clear sky'],1:['🌤','Mainly clear'],2:['⛅','Partly cloudy'],3:['☁️','Overcast'],
  45:['🌫','Fog'],48:['🌫','Depositing rime fog'],
  51:['🌦','Light drizzle'],53:['🌦','Drizzle'],55:['🌧','Dense drizzle'],
  61:['🌧','Slight rain'],63:['🌧','Rain'],65:['🌧','Heavy rain'],
  71:['🌨','Slight snow'],73:['🌨','Snow'],75:['❄️','Heavy snow'],
  80:['🌦','Rain showers'],81:['🌧','Rain showers'],82:['⛈','Violent showers'],
  95:['⛈','Thunderstorm'],96:['⛈','Thunderstorm w/ hail'],99:['⛈','Severe thunderstorm']
};
function wmoInfo(code){ return WMO[code] || ['🌡','Unknown']; }

async function geocode(name){
  const url = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(name)}&count=1`;
  const res = await fetch(url);
  const data = await res.json();
  if(!data.results || !data.results.length) throw new Error('Location not found');
  const r = data.results[0];
  return { lat:r.latitude, lon:r.longitude, label:`${r.name}${r.admin1 ? ', '+r.admin1:''}` };
}

async function fetchWeather(lat, lon){
  const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}` +
    `&current=temperature_2m,relative_humidity_2m,apparent_temperature,weather_code,wind_speed_10m` +
    `&hourly=temperature_2m,weather_code` +
    `&daily=weather_code,temperature_2m_max,temperature_2m_min` +
    `&timezone=auto&forecast_days=7`;
  const res = await fetch(url);
  if(!res.ok) throw new Error('Weather API error');
  return res.json(); // <-- this is the JSON payload we render below
}

function renderWeather(json, label){
  const cur = json.current;
  const [curIcon, curDesc] = wmoInfo(cur.weather_code);

  document.getElementById('placeName').textContent = label;
  document.getElementById('curTemp').textContent = Math.round(cur.temperature_2m);
  document.getElementById('curDesc').textContent = curDesc;
  document.getElementById('curIcon').textContent = curIcon;
  document.getElementById('feelsLike').textContent = Math.round(cur.apparent_temperature) + '°';
  document.getElementById('humidity').textContent = cur.relative_humidity_2m + '%';
  document.getElementById('wind').textContent = Math.round(cur.wind_speed_10m) + ' km/h';

  // Hourly: next 24 hours from now
  const hourly = json.hourly;
  const nowIdx = hourly.time.findIndex(t => t === json.current.time.slice(0,13)+':00') ;
  const startIdx = nowIdx >= 0 ? nowIdx : 0;
  const hourlyEl = document.getElementById('hourly');
  hourlyEl.innerHTML = '';
  for(let i = startIdx; i < startIdx + 24 && i < hourly.time.length; i++){
    const dt = new Date(hourly.time[i]);
    const hourLabel = dt.getHours().toString().padStart(2,'0') + ':00';
    const [ic] = wmoInfo(hourly.weather_code[i]);
    const div = document.createElement('div');
    div.className = 'hour-card';
    div.innerHTML = `<div class="t">${hourLabel}</div><div class="ic">${ic}</div><div class="v">${Math.round(hourly.temperature_2m[i])}°</div>`;
    hourlyEl.appendChild(div);
  }

  // Daily
  const daily = json.daily;
  const dailyEl = document.getElementById('daily');
  dailyEl.innerHTML = '';
  const dayNames = ['Sun','Mon','Tue','Wed','Thu','Fri','Sat'];
  daily.time.forEach((d, i) => {
    const dt = new Date(d);
    const name = i === 0 ? 'Today' : dayNames[dt.getDay()];
    const [ic] = wmoInfo(daily.weather_code[i]);
    const row = document.createElement('div');
    row.className = 'day-row';
    row.innerHTML = `<div class="dname">${name}</div><div class="dic">${ic}</div>
      <div class="drange">${Math.round(daily.temperature_2m_min[i])}° / ${Math.round(daily.temperature_2m_max[i])}°</div>`;
    dailyEl.appendChild(row);
  });

  statusEl.style.display = 'none';
  contentEl.style.display = 'block';
}

async function loadByName(name){
  try{
    statusEl.style.display = 'block';
    statusEl.textContent = 'Loading weather…';
    contentEl.style.display = 'none';
    const loc = await geocode(name);
    const json = await fetchWeather(loc.lat, loc.lon);
    renderWeather(json, loc.label);
  }catch(err){
    statusEl.textContent = 'Could not load weather: ' + err.message;
  }
}

async function loadByCoords(lat, lon, label){
  try{
    statusEl.style.display = 'block';
    statusEl.textContent = 'Loading weather…';
    contentEl.style.display = 'none';
    const json = await fetchWeather(lat, lon);
    renderWeather(json, label || `${lat.toFixed(2)}, ${lon.toFixed(2)}`);
  }catch(err){
    statusEl.textContent = 'Could not load weather: ' + err.message;
  }
}

document.getElementById('locForm').addEventListener('submit', e => {
  e.preventDefault();
  const val = document.getElementById('locInput').value.trim();
  if(val) loadByName(val);
});

document.getElementById('geoBtn').addEventListener('click', () => {
  if(!navigator.geolocation){
    statusEl.textContent = 'Geolocation not supported by this browser.';
    return;
  }
  statusEl.style.display = 'block';
  statusEl.textContent = 'Getting your location…';
  navigator.geolocation.getCurrentPosition(
    pos => loadByCoords(pos.coords.latitude, pos.coords.longitude, 'Your location'),
    () => { statusEl.textContent = 'Location access denied. Search a city instead.'; }
  );
});

// Initial load: Salem, Tamil Nadu
loadByName('Salem, Tamil Nadu');
</script>
